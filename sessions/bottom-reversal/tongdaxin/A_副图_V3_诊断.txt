{====== 底部横盘·仙人指路·1~3日反包 副图诊断 V3 ======}
{说明：技术指标公式（副图）。必须整段粘贴，不能只贴“副图输出”一段，否则报“未知字符串 HARD”。
 坐标显示不对时，在指标属性里把坐标固定为 0~170。}
M:=60; N:=180; BOX_MAX:=28;
WIN:=3;
FB_MODE:=1;
SC_MIN:=55;
DP:=0;

ZF:=IF(CODELIKE('30') OR CODELIKE('68'),0.2,
    IF(CODELIKE('8') OR CODELIKE('4') OR CODELIKE('92'),0.3,0.1));
K:=IF(ZF=0.1,1,IF(ZF=0.2,1.6,2));
KS:=IF(K>1,1.5,1);
JC:=NOT(NAMEINCLUDE('ST') OR NAMEINCLUDE('退'))
    AND BARSCOUNT(C)>=N+10
    AND MA(AMOUNT,5)>30000000 AND V>0
    AND (DP=0 OR INDEXC>=MA(INDEXC,20));

VB:=MAX(REF(MA(V,40),5),1);
ATRP:=REF(MA(TR,20),1)/MAX(REF(C,1),0.01)*100;
UPS:=H-MAX(O,C);
SD:=UPS/MAX(REF(C,1),0.01)*100>=MAX(2.0,ATRP*0.5)
    AND UPS/MAX(H-L,0.01)>=0.40
    AND C>=REF(C,1)*(1-0.03*K)
    AND V>=VB*1.1 AND V<=VB*6;
YD_A:=V>=VB*1.6 AND C>O AND C/REF(C,1)>=1+0.025*K
      AND (C-L)/MAX(H-L,0.01)>=0.55;
YD_B:=V>=VB*1.5 AND H/REF(C,1)>=1+0.03*K AND C>=REF(C,1)*0.99;
LJYD:=YD_A OR YD_B;
CH_BAD:=C<REF(C,1)*(1-0.05*K) AND V>=VB*1.5;

D:=REF(BARSLAST(SD),1)+1;
SD_OK:=D>=1 AND D<=WIN;
B:=D+1;
SH:=REF(H,D);
SH_TOP:=REF(MAX(O,C),D);
SH_UP:=SH-SH_TOP;
NOBREAK:=LLV(L,D)>=REF(L,D)*0.99;
FIRST:=REF(HHV(C,D),1)<SH;

HH:=REF(HHV(H,N),B); LL:=REF(LLV(L,N),B);
RG:=MAX(HH-LL,0.01);
POS_B:=(REF(C,B)-LL)/RG*100;
POS_T:=(C-LL)/RG*100;
IS_BOT:=POS_B<=50 AND POS_T<=60 AND HH/MAX(LL,0.01)>=1.15;

H55:=REF(HHV(C,M-5),B+3);
L55:=REF(LLV(C,M-5),B+3);
AMP_BOX:=(H55/MAX(L55,0.01)-1)*100;
BOX_LIM:=BOX_MAX*IF(K>1,1.25,1);
BOX_OK:=AMP_BOX<=BOX_LIM
        AND (REF(C,B)/MAX(REF(LLV(L,M),B),0.01)-1)*100<=BOX_LIM
        AND REF(C,B)<=H55*1.08;
FLAT60:=ABS(REF(MA(C,60),B)/MAX(REF(MA(C,60),B+20),0.01)-1)*100<=10;

V_BASE:=MAX(REF(MA(V,40),B+3),1);
UP_V:=REF(SUM(IF(C>O OR (C=O AND C>=REF(C,1)),V,0),10),B);
DN_V:=REF(SUM(IF(C<O OR (C=O AND C<REF(C,1)),V,0),10),B);
YY_RATIO:=UP_V/MAX(DN_V,1);
DUI_V:=REF(MA(V,5),B)/V_BASE;
NYD:=REF(COUNT(LJYD,20),B);
NBAD:=REF(COUNT(CH_BAD,10),B);

T_YANG:=C>O AND C>REF(C,1) AND NOT(O=H AND H=L)
        AND (C-O)/MAX(H-L,0.01)>=0.40
        AND (H-C)/C*100<=2.5*KS
        AND V>=V_BASE*1.1;
FB_1:=C>=SH;
FB_2:=H>=SH AND C>=SH_TOP+SH_UP*0.70;
T_FANBAO:=IF(FB_MODE=1,FB_1,FB_2);
NO_CHASE:=C/REF(C,5)<=1+0.2*K AND C>MA(C,20);

S1:=IF(BOX_OK,20,IF(AMP_BOX<=BOX_LIM*1.3,10,0));
S2:=IF(FLAT60,10,0);
S3:=IF(DUI_V>=1.2 OR NYD>=2,15,IF(DUI_V>=1.05 OR NYD>=1,8,0));
S4:=IF(YY_RATIO>=1.3,10,IF(YY_RATIO>=1.1,5,0));
S5:=IF(NBAD=0,5,0);
S6:=IF(SH>=H55*0.97,10,0);
S7:=IF(C>H55,10,0);
S8:=IF(V>=V_BASE*1.6,10,IF(V>=V_BASE*1.2,5,0));
S9:=IF((C-O)/MAX(H-L,0.01)>=0.6 AND (H-C)/C*100<=1.5*KS,10,0);
SCORE:=S1+S2+S3+S4+S5+S6+S7+S8+S9;

HARD:=JC AND SD_OK AND IS_BOT AND NOBREAK AND FIRST
      AND T_YANG AND T_FANBAO AND NO_CHASE;

XG1:=HARD AND SCORE>=SC_MIN;
SC:=IF(SD_OK,SCORE,0);
FB_ALL:=SD_OK AND T_YANG AND T_FANBAO AND NOBREAK AND FIRST;

STICKLINE(HARD,0,SC,0.8,0),COLORRED;
STICKLINE(SD_OK AND NOT(HARD),0,SC,0.8,1),COLORGRAY;
门槛:SC_MIN,COLORYELLOW,DOTLINE;
DRAWTEXT(XG1,SC+6,'▲反包'),COLORMAGENTA;

STICKLINE(JC,110,116,3,0),COLORGRAY;
STICKLINE(SD_OK AND IS_BOT,120,126,3,0),COLORWHITE;
STICKLINE(SD_OK AND BOX_OK,130,136,3,0),COLORCYAN;
STICKLINE(SD_OK AND S3+S4>=15,140,146,3,0),COLORYELLOW;
STICKLINE(SD,150,156,3,0),COLORGREEN;
STICKLINE(FB_ALL,160,166,3,0),COLORRED;
DRAWTEXT(ISLASTBAR,111,' 基础'),COLORGRAY;
DRAWTEXT(ISLASTBAR,121,' 底部'),COLORWHITE;
DRAWTEXT(ISLASTBAR,131,' 横盘'),COLORCYAN;
DRAWTEXT(ISLASTBAR,141,' 资金'),COLORYELLOW;
DRAWTEXT(ISLASTBAR,151,' 试盘'),COLORGREEN;
DRAWTEXT(ISLASTBAR,161,' 反包'),COLORRED;

评分:SC,NODRAW;
硬条件:HARD,NODRAW;
底部位置:POS_B,NODRAW;
箱体振幅:AMP_BOX,NODRAW;
距上影:IF(SD_OK,D,0),NODRAW;
上界:170,NODRAW;
